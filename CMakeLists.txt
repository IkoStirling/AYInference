cmake_minimum_required(VERSION 3.25)
project(AYInference VERSION 0.3.0 LANGUAGES CXX)
if(NOT TARGET AYNetworkHttp OR NOT TARGET AYSerializer)
    message(FATAL_ERROR "AYInference requires AYNetworkHttp and AYSerializer")
endif()
find_package(nlohmann_json CONFIG REQUIRED)
add_library(AYInference STATIC src/Inference.cpp)
target_compile_features(AYInference PUBLIC cxx_std_23)
target_include_directories(AYInference PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(AYInference PUBLIC AYSerializer PRIVATE AYNetworkHttp nlohmann_json::nlohmann_json)
if(BUILD_TESTING AND TARGET AYTest)
    find_package(httplib CONFIG REQUIRED)
    add_executable(AYInference_Tests unittest/Test_Inference.cpp unittest/Test_NativeTools.cpp)
    target_link_libraries(AYInference_Tests PRIVATE AYInference AYNetworkHttp AYTest httplib::httplib nlohmann_json::nlohmann_json)
    ay_add_test(AYInference_Tests CASE_REPORT TIMEOUT 30 LABELS "inference;services;unit")
endif()
